private theorem graphIdealClass_fieldBaseChange
    (K : Type u) [Field K] [Algebra k K]
    (T : FScheme K) (s : T ⟶ fieldBaseChange K X) :
    let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
    letI := hX.smooth
    letI := hX.proper
    letI : SmoothOfRelativeDimension 1 (pullback.snd X.hom b) :=
      (curveProperties_baseChange hX K b).smooth
    letI : IsProper (pullback.snd X.hom b) := (curveProperties_baseChange hX K b).proper
    pullbackClass (familyBaseChangeIso X.hom b T).hom
        (classOf _ (Hom.graphIdealLineBundle Hom.smoothRelativeDivisorIdeal
          X.hom (T.hom ≫ b) (s.left ≫ pullback.fst X.hom b)
          (by
            rw [Category.assoc, pullback.condition, ← Category.assoc]
            exact congrArg (fun v => v ≫ b)
              (show s.left ≫ pullback.snd X.hom b = T.hom from s.w)))) =
      classOf _ (Hom.graphIdealLineBundle Hom.smoothRelativeDivisorIdeal
        (pullback.snd X.hom b) T.hom s.left s.w) := by
  let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
  let := hX.smooth
  let := hX.proper
  let : SmoothOfRelativeDimension 1 (pullback.snd X.hom b) :=
    (curveProperties_baseChange hX K b).smooth
  let : IsProper (pullback.snd X.hom b) := (curveProperties_baseChange hX K b).proper
  let e := familyBaseChangeIso X.hom b T
  let Y := pullback X.hom (T.hom ≫ b)
  let Y' := pullback (pullback.snd X.hom b) T.hom
  let i : T.left ⟶ Y := Hom.graph X.hom (T.hom ≫ b)
    (s.left ≫ pullback.fst X.hom b)
    (by
      rw [Category.assoc, pullback.condition, ← Category.assoc]
      exact congrArg (fun v => v ≫ b)
        (show s.left ≫ pullback.snd X.hom b = T.hom from s.w))
  let i' : T.left ⟶ Y' := Hom.graph (pullback.snd X.hom b) T.hom s.left s.w
  have he : i' ≫ e.hom = i := by
    apply pullback.hom_ext
    · simp only [Category.assoc, e, familyBaseChangeIso_hom_fst,
        i, i', Hom.graph_fst, Hom.graph_fst_assoc]
    · simp only [Category.assoc, e, familyBaseChangeIso_hom_snd,
        i, i', Hom.graph_snd]
  have h : (𝟙 T.left) ≫ i = i' ≫ e.hom := (Category.id_comp i).trans he.symm
  have hfamily : IsPullback e.hom
      (pullback.snd (pullback.snd X.hom b) T.hom)
      (pullback.snd X.hom (T.hom ≫ b)) (𝟙 T.left) :=
    IsPullback.of_horiz_isIso ⟨by
      simpa only [Category.comp_id, e] using familyBaseChangeIso_hom_snd X.hom b T⟩
  have hgraph : IsPullback (𝟙 T.left) i' i e.hom := IsPullback.of_horiz_isIso ⟨h⟩
  have hi : IsClosedImmersion i := by dsimp [i]; infer_instance
  have hrank : IsLocallyRankOne Y (Hom.idealModule i) :=
    Hom.graphIdeal_isLocallyRankOne Hom.smoothRelativeDivisorIdeal X.hom (T.hom ≫ b)
      (s.left ≫ pullback.fst X.hom b) _
  have hflat : Flat (i ≫ pullback.snd X.hom (T.hom ≫ b)) := by
    dsimp [i]
    rw [Hom.graph_snd]
    infer_instance
  have hIso : IsIso (Hom.idealPullbackMap i i' e.hom (𝟙 T.left) h) :=
    Hom.relativeCartierIdealBaseChange T.left T.left Y Y' T.left T.left
      (pullback.snd X.hom (T.hom ≫ b)) (𝟙 T.left)
      (pullback.snd (pullback.snd X.hom b) T.hom) e.hom i i' (𝟙 T.left) h
      hfamily hgraph hi hrank hflat
  let := hIso
  exact Quotient.sound ⟨asIso (Hom.idealPullbackMap i i' e.hom (𝟙 T.left) h)⟩

theorem positiveGraphClass_fieldBaseChange
    (K : Type u) [Field K] [Algebra k K]
    (T : FScheme K) (s : T ⟶ fieldBaseChange K X) :
    let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
    letI := hX.smooth
    letI := hX.proper
    letI : SmoothOfRelativeDimension 1 (pullback.snd X.hom b) :=
      (curveProperties_baseChange hX K b).smooth
    letI : IsProper (pullback.snd X.hom b) := (curveProperties_baseChange hX K b).proper
    pullbackClass (familyBaseChangeIso X.hom b T).hom
        (classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
          X.hom (T.hom ≫ b) (s.left ≫ pullback.fst X.hom b)
          (by
            rw [Category.assoc, pullback.condition, ← Category.assoc]
            exact congrArg (fun v => v ≫ b)
              (show s.left ≫ pullback.snd X.hom b = T.hom from s.w)))) =
      classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
        (pullback.snd X.hom b) T.hom s.left s.w) := by
  let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
  let := hX.smooth
  let := hX.proper
  let : SmoothOfRelativeDimension 1 (pullback.snd X.hom b) :=
    (curveProperties_baseChange hX K b).smooth
  let : IsProper (pullback.snd X.hom b) := (curveProperties_baseChange hX K b).proper

      (-classOf _ (Hom.graphIdealLineBundle Hom.smoothRelativeDivisorIdeal
        X.hom (T.hom ≫ b) (s.left ≫ pullback.fst X.hom b) _)) =
    -classOf _ (Hom.graphIdealLineBundle Hom.smoothRelativeDivisorIdeal
      (pullback.snd X.hom b) T.hom s.left s.w)
  rw [← PullbackTensor.Local.pullbackHom_apply, map_neg,
    PullbackTensor.Local.pullbackHom_apply, graphIdealClass_fieldBaseChange hX K T s]

private theorem homEquiv_comp_positiveCurveMorphism
    {T : LocallyNoetherianOver (Spec (.of k))} (s : T.obj ⟶ X) :
    letI := hX.smooth
    letI := hX.proper
    R.homEquiv (ObjectProperty.homMk
        (s ≫ positiveCurveMorphism hX P R) : T ⟶ P) =
      ((quotientProjection X.hom ≫ toFppfPicardSheaf X.hom).app (op T.obj)).hom
        (classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
          X.hom T.obj.hom s.left s.w)) := by
  let := hX.smooth
  let := hX.proper
  let a : T ⟶ locallyNoetherianParameter X 1 := ObjectProperty.homMk
    (Over.homMk (parameterPoint X.hom T.obj.hom (fun _ : Fin 1 => s.left) (fun _ => s.w))
      (parameterPoint_over X.hom T.obj.hom _ _))
  have he : (ObjectProperty.homMk
      (s ≫ positiveCurveMorphism hX P R) : T ⟶ P) =
      a ≫ positiveAbelMorphism X P R 1 := by
    ext : 1
    change s ≫ (singleParameterMorphism X ≫ (positiveAbelMorphism X P R 1).hom) = _
    rw [← Category.assoc, comp_singleParameterMorphism]
    rfl
  have hclass : pullbackClass ((familyScheme X.hom).map a.hom)
      (classOf _ (positiveFamilyLineBundle X.hom 1)) =
      classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
        X.hom T.obj.hom s.left s.w) := by
    have hm : (familyScheme X.hom).map a.hom =
        Hom.familyMap X.hom (parameter X.hom 1).hom T.obj.hom
          (parameterPoint X.hom T.obj.hom (fun _ : Fin 1 => s.left) (fun _ => s.w))
          (parameterPoint_over X.hom T.obj.hom _ _) :=
      (RelativePicard.Classes.familyMap_eq_familyScheme_map X.hom a.hom).symm
    rw [hm]
    simpa only [Fin.sum_univ_one, familyScheme_obj, locallyNoetherianParameter] using
      positiveFamilyLineBundle_class_at_tuple X.hom T.obj.hom
        (fun _ : Fin 1 => s.left) (fun _ => s.w)
  exact (congrArg R.homEquiv he).trans
    ((homEquiv_comp_positiveAbelMorphism X P R a).trans
      (congrArg ((quotientProjection X.hom ≫ toFppfPicardSheaf X.hom).app (op T.obj)).hom
        hclass))

/-- The original positive degree-one classifier commutes with field
extension on the SAME base-changed Picard representative. -/
theorem positiveCurveMorphism_baseChange
    (K : Type u) [Field K] [Algebra k K] :
    let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
    positiveCurveMorphism (curveProperties_baseChange hX K b)
        (LocallyNoetherianOver.baseChange b P) (baseChangeRepresentableBy X.hom b P R) =
      (Over.pullback b).map (positiveCurveMorphism hX P R) := by
  let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
  let X' := fieldBaseChange K X
  let hX' := curveProperties_baseChange hX K b
  let := hX.smooth
  let := hX.proper
  let := hX'.smooth
  let := hX'.proper
  let P' := LocallyNoetherianOver.baseChange b P
  let R' := baseChangeRepresentableBy X.hom b P R
  let T : LocallyNoetherianOver (Spec (.of K)) :=
    ⟨X', LocallyOfFiniteType.isLocallyNoetherian X'.hom⟩
  let T0 := (LocallyNoetherianOver.map b).obj T
  let q : T0.obj ⟶ X :=
    ((Over.mapPullbackAdj b).homEquiv T.obj X).symm (𝟙 X')
  have hqfst : q.left = pullback.fst X.hom b := by
    change (𝟙 (pullback X.hom b)) ≫ pullback.fst X.hom b = _
    exact Category.id_comp _
  have hq : ((Over.mapPullbackAdj b).homEquiv T.obj P.obj).symm
      ((Over.pullback b).map (positiveCurveMorphism hX P R)) =
      q ≫ positiveCurveMorphism hX P R := by
    simpa only [Category.id_comp] using
      (Over.mapPullbackAdj b).homEquiv_naturality_right_symm
        (𝟙 X') (positiveCurveMorphism hX P R)
  let c : PicardGroup (pullback X.hom T0.obj.hom) :=
    classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
      X.hom T0.obj.hom q.left q.w)
  let c' : PicardGroup (pullback X'.hom X'.hom) :=
    classOf _ (Hom.positiveGraphLineBundle Hom.smoothRelativeDivisorIdeal
      X'.hom X'.hom (𝟙 X'.left) (Category.id_comp _))
  have hc : pullbackClass (familyBaseChangeIso X.hom b X').hom c = c' := by
    have H := positiveGraphClass_fieldBaseChange hX K X' (𝟙 X')
    simpa only [Over.id_left, Category.id_comp, hqfst, c, c', T0, T, X', b,
      LocallyNoetherianOver.map_obj_obj, Over.map_obj_hom, fieldBaseChange, Over.mk_hom] using H
  have hc' : pullbackClass (familyBaseChangeIso X.hom b X').inv c' = c := by
    rw [← hc, ← pullbackClass_comp, Iso.inv_hom_id, pullbackClass_id]
  let q0 : PicardGroup (pullback X.hom T0.obj.hom) →+
      (fppfPicardSheaf X.hom).obj.obj (op T0.obj) :=
    ((quotientProjection X.hom ≫ toFppfPicardSheaf X.hom).app (op T0.obj)).hom
  let q1 : PicardGroup (pullback X'.hom X'.hom) →+
      (fppfPicardSheaf X'.hom).obj.obj (op X') :=
    ((quotientProjection X'.hom ≫ toFppfPicardSheaf X'.hom).app (op X')).hom
  have hleft : ((fppfPicardSheafBaseChangeIso X.hom b).hom.app (op X')).hom
      (R'.homEquiv (ObjectProperty.homMk
        (positiveCurveMorphism hX' P' R') : T ⟶ P')) = q0 c := by
    have H := homEquiv_comp_positiveCurveMorphism hX' P' R' (T := T) (𝟙 X')
    rw [Category.id_comp] at H
    change R'.homEquiv (ObjectProperty.homMk
      (positiveCurveMorphism hX' P' R') : T ⟶ P') = q1 c' at H
    rw [H]
    have Hbase := fppfPicardSheafBaseChangeIso_hom_app_toFppfPicardSheaf_mk
      X.hom b (op X') c'
    change ((fppfPicardSheafBaseChangeIso X.hom b).hom.app (op X')).hom (q1 c') =
      q0 (pullbackClass (familyBaseChangeIso X.hom b X').inv c') at Hbase
    exact Hbase.trans (congrArg q0 hc')
  have hright : ((fppfPicardSheafBaseChangeIso X.hom b).hom.app (op X')).hom
      (R'.homEquiv (ObjectProperty.homMk
        ((Over.pullback b).map (positiveCurveMorphism hX P R)) : T ⟶ P')) = q0 c := by
    have H := baseChangeRepresentableBy_homEquiv X.hom b P R (T := T)
      (ObjectProperty.homMk ((Over.pullback b).map (positiveCurveMorphism hX P R)) : T ⟶ P')
    change R'.homEquiv (ObjectProperty.homMk
      ((Over.pullback b).map (positiveCurveMorphism hX P R)) : T ⟶ P') =
      ((fppfPicardSheafBaseChangeIso X.hom b).inv.app (op X')).hom
        (R.homEquiv (ObjectProperty.homMk
          (((Over.mapPullbackAdj b).homEquiv T.obj P.obj).symm
            ((Over.pullback b).map (positiveCurveMorphism hX P R))) : T0 ⟶ P)) at H
    rw [hq] at H
    have Hclass := homEquiv_comp_positiveCurveMorphism hX P R (T := T0) q
    change R.homEquiv (ObjectProperty.homMk (q ≫ positiveCurveMorphism hX P R) : T0 ⟶ P) =
      q0 c at Hclass
    rw [Hclass] at H
    rw [H]
    exact fppfPicardSheafBaseChangeIso_hom_app_inv_app_apply X.hom b (q0 c)
  have H : (ObjectProperty.homMk (positiveCurveMorphism hX' P' R') : T ⟶ P') =
      ObjectProperty.homMk ((Over.pullback b).map (positiveCurveMorphism hX P R)) := by
    apply R'.homEquiv.injective
    apply ((fppfPicardSheafBaseChangeIso X.hom b).app (op X')).addCommGroupIsoToAddEquiv.injective
    exact hleft.trans hright.symm
  exact congrArg (fun v : T ⟶ P' => v.hom) H

    (M : GrpObj P.obj) (hM : M = (picardCommGrpObj X.hom P R).toGrpObj)

    letI : DivisionMonoid (T ⟶ P.obj) := Group.toDivisionMonoid

      (x ≫ positiveCurveMorphism hX P R) / (y ≫ positiveCurveMorphism hX P R) := by

  let : DivisionMonoid (T ⟶ P.obj) := Group.toDivisionMonoid

private theorem hom_div_of_grpObj_eq
    {C : Type*} [Category C] [CartesianMonoidalCategory C]
    (G : C) (M N : GrpObj G) (h : M = N)
    {T : C} (a b : T ⟶ G) :
    (letI : GrpObj G := M; a / b) = (letI : GrpObj G := N; a / b) := by
  cases h
  rfl

private theorem adjunction_homEquiv_div_of_grpObj_eq
    {C D : Type*} [Category C] [Category D]
    [CartesianMonoidalCategory C] [CartesianMonoidalCategory D]
    {L : C ⥤ D} {F : D ⥤ C} (adj : L ⊣ F) [F.Monoidal]
    (G : D) [hG : GrpObj G] (native : GrpObj (F.obj G))
    (h : native = Functor.grpObjObj (F := F) (G := G))
    (T : C) (a c : L.obj T ⟶ G) :
    letI : GrpObj (F.obj G) := native
    adj.homEquiv T G (a / c) = (adj.homEquiv T G a) / (adj.homEquiv T G c) := by
  let N : GrpObj (F.obj G) := Functor.grpObjObj (F := F) (G := G)
  have hdiv := hom_div_of_grpObj_eq (F.obj G) native N h
    (adj.homEquiv T G a) (adj.homEquiv T G c)
  let : GrpObj (F.obj G) := Functor.grpObjObj (F := F) (G := G)
  let : Group (T ⟶ F.obj G) := CategoryTheory.Hom.group (G := F.obj G) (X := T)
  let : DivisionMonoid (T ⟶ F.obj G) := Group.toDivisionMonoid
  have hAdj := (adj.homMulEquiv T).map_div a c
  simp only [Adjunction.homMulEquiv_apply] at hAdj
  have hfinal := hAdj.trans hdiv.symm
  exact hfinal

/-- The original difference scalar comparison on each locally Noetherian
scheme test, including nonreduced tests. The original R is used only on
that actual test and its original restriction of scalars. -/
theorem jacobianDifferenceMorphism_baseChange
    (K : Type u) [Field K] [Algebra k K]
    {T : LocallyNoetherianOver (Spec (.of K))}
    (x y : T.obj ⟶ Over.mk
      (pullback.snd X.hom (Spec.map (CommRingCat.ofHom (algebraMap k K))))) :
    let b := Spec.map (CommRingCat.ofHom (algebraMap k K))

/-- The SAME origin-free canonical map is an isomorphism over the original
field. The coherent H1 inequality remains the original explicit premise. -/
theorem isIso_canonicalAbelDualMorphism
    (hH1 : letI := picardCommGrpObj X.hom P R
      Module.finrank k (structureCohomology k (picardIdentityComponent X.hom P R) 1) ≤
        Module.finrank k (Cotangent.Space η[picardIdentityComponent X.hom P R])) :
    IsIso (canonicalAbelDualMorphism hX P R Q RQ hH1) := by
  let := picardCommGrpObj X.hom P R
  let K := AlgebraicClosure k
  let b := Spec.map (CommRingCat.ofHom (algebraMap k K))
  let F := Over.pullback b
  let X' := fieldBaseChange K X
  let hX' := curveProperties_baseChange hX K b
  let := hX'.smooth
  let := hX'.proper
  let : GeometricallyIntegral X'.hom := hX'.isGeometricallyIntegral
  have : Surjective X'.hom := inferInstance
  obtain ⟨o, ho⟩ := Scheme.Hom.exists_lift_spec_of_isAlgClosed X'.hom K (𝟙 _)
  let u : 𝟙_ (FScheme K) ⟶ X' := Over.homMk o ho
  let J := picardIdentityComponent X.hom P R
  let D := picardIdentityComponent J.hom Q RQ
  let A := fieldBaseChange K J
  let Q' := LocallyNoetherianOver.baseChange b Q
  let RQA := baseChangeRepresentableBy J.hom b Q RQ
  let e := jacobianScalarExtensionIso P R K
  let eD : fieldBaseChange K D ≅ picardIdentityComponent A.hom Q' RQA :=
    Over.isoMk (picardIdentityComponentBaseChangeIso J.hom Q RQ K)
      (picardIdentityComponentBaseChangeIso_hom_comp J.hom Q RQ K)
  let m := canonicalAbelDualMorphism hX P R Q RQ hH1
  have H := canonicalAbelDualMorphism_fieldComparison hX P R Q RQ hH1 K u
  let := isIso_geometricAbelPullback hX P R Q RQ hH1 K u
  have : IsIso (F.map m ≫ e.hom) := by
    rw [H]
    infer_instance
  let : IsIso (F.map m) := IsIso.of_isIso_comp_right (F.map m) e.hom
  let : Surjective b := ⟨Function.surjective_to_subsingleton b⟩
  have hmap : IsIso (F.map m).left :=
    inferInstanceAs (IsIso ((Over.forget (Spec (.of K))).map (F.map m)))
  have hSquare : IsPullback (F.map m).left (pullback.fst D.hom b)
      (pullback.fst J.hom b) m.left := by
    apply (IsPullback.paste_horiz_iff (IsPullback.of_hasPullback J.hom b).flip
      (show (F.map m).left ≫ pullback.fst J.hom b =
        pullback.fst D.hom b ≫ m.left from pullback.lift_fst _ _ _)).mp
    have hSnd : (F.map m).left ≫ pullback.snd J.hom b = pullback.snd D.hom b :=
      (F.map m).w
    have hStructure : m.left ≫ J.hom = D.hom := m.w
    rw [hSnd, hStructure]
    exact (IsPullback.of_hasPullback D.hom b).flip
  have hbase : (@Surjective ⊓ @Flat ⊓ @QuasiCompact : MorphismProperty Scheme.{u})
      (pullback.fst J.hom b) := ⟨⟨inferInstance, inferInstance⟩, inferInstance⟩
  have : IsIso m.left := MorphismProperty.of_isPullback_of_descendsAlong
    (P := MorphismProperty.isomorphisms Scheme) hSquare hbase hmap
  let : IsIso ((Over.forget (Spec (.of k))).map m) := this
  exact isIso_of_reflects_iso m (Over.forget (Spec (.of k)))
