attribute [local instance] Over.cartesianMonoidalCategory

variable {k : Type u} [Field k] (X Y : FScheme k)

open KunnethOpenMayerVietoris

namespace KunnethOpenMayerVietoris

/-- The original coordinate restriction retains every original antidiagonal generator. -/
theorem structureOpenKunnethRestriction_lof {U V : Y.left.Opens} (i : U ⟶ V)
    (n : ℕ) (s : HomologicalComplex₂.TotalDegree n)
    (z : structureCohomology k X s.val.1 ⊗[k] structureOpenCohomology Y V s.val.2) :
    structureOpenKunnethRestriction X Y i n (DirectSum.lof k _ _ s z) =
      DirectSum.lof k _ _ s
        (TensorProduct.map LinearMap.id (structureOpenRestriction Y i s.val.2) z) := by
  unfold structureOpenKunnethRestriction Kunneth.MayerVietoris.tensorMap
  exact DirectSum.lmap_lof _ s z

end KunnethOpenMayerVietoris

end AlgebraicGeometry.Scheme.Cohomology
